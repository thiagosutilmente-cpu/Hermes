        const minFareInput = document.getElementById('setMinFare');
        const minKmInput = document.getElementById('setMinKm');
        if (!minFareInput || !minKmInput) return;

        const currentMinFare = parseFloat(minFareInput.value) || parseFloat(currentDriverSettings.minFareValue) || 8.0;
        const currentMinKm = parseFloat(minKmInput.value) || parseFloat(currentDriverSettings.minValuePerKm) || 1.5;
        
        const newMinFare = (currentMinFare * 1.3).toFixed(2);
        const newMinKm = (currentMinKm * 1.3).toFixed(2);
        
        minFareInput.value = newMinFare;
        minKmInput.value = newMinKm;
        
        if (window.currentDriverSettings) {
          window.currentDriverSettings.minFareValue = parseFloat(newMinFare);
          window.currentDriverSettings.minValuePerKm = parseFloat(newMinKm);
        }
        
        if (currentDriverId && window.saveDriverSettings) {
          await window.saveDriverSettings(currentDriverId, {
            minFareValue: parseFloat(newMinFare),
            minValuePerKm: parseFloat(newMinKm)
          });
        }
        
        const wWarn = document.getElementById('weatherWarning');
        if (wWarn) wWarn.style.display = 'none';
        
        if (window.showToast) window.showToast('Modo Chuva ativado: Valores mínimos aumentados em 30%.', 'success');
        if (window.speakText) window.speakText('Modo chuva ativado, Thiago. Os filtros de valor mínimo foram aumentados em trinta por cento.');
        if (window.triggerHapticFeedback) window.triggerHapticFeedback('success');
    };
